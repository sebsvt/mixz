function goPage(number) {

  document.querySelectorAll(".screen")
    .forEach(page => page.classList.remove("active"));

  document.getElementById("page" + number)
    .classList.add("active");

  window.scrollTo({
    top: 0,
    behavior: "smooth"
  });

  if(number === 4) {

    const message = CONFIG.birthdayMessage
      .replaceAll("ที่รัก", CONFIG.loverName);

    document.getElementById("letterMessage")
      .textContent = message;

    document.getElementById("senderDisplay")
      .textContent = CONFIG.senderName + " ♡";

  }

}

// PASSWORD: 15/8/2569

const CORRECT_PASSWORD = "15082569";

let enteredPassword = "";

function updateKeypadDisplay() {

  const display = document.getElementById("keypadDisplay");

  const day = enteredPassword.slice(0,2);
  const month = enteredPassword.slice(2,4);
  const year = enteredPassword.slice(4,8);

  const mask = value =>
    "●".repeat(value.length);

  const dayDisplay = day.length ? mask(day) : "DD";
  const monthDisplay = month.length ? mask(month) : "MM";
  const yearDisplay = year.length ? mask(year) : "YYYY";

  display.textContent =
    `${dayDisplay} / ${monthDisplay} / ${yearDisplay}`;

}

function keypadPress(number) {

  if(enteredPassword.length >= 8) return;

  enteredPassword += number;

  updateKeypadDisplay();

  document.getElementById("keypadError")
    .textContent = "";

}

function keypadDelete() {

  enteredPassword = enteredPassword.slice(0,-1);

  updateKeypadDisplay();

}

function keypadClear() {

  enteredPassword = "";

  updateKeypadDisplay();

  document.getElementById("keypadError")
    .textContent = "";

}

function keypadSubmit() {

  const display = document.getElementById("keypadDisplay");

  const error = document.getElementById("keypadError");

  if(enteredPassword.length !== 8) {

    error.textContent =
      "ใส่รหัสให้ครบ 8 หลักก่อนนะะ ♡";

    display.classList.remove("shake");

    void display.offsetWidth;

    display.classList.add("shake");

    return;

  }

  if(enteredPassword === CORRECT_PASSWORD) {

    error.textContent = "เก่งมากกไปเป่าเค้กกัน ♡";

    confetti();

    setTimeout(() => {

      keypadClear();

      goPage(3);

    }, 1200);

  } else {

    error.textContent =
      "รหัสยังไม่ถูกน้า เค้างอนเเล้วว 🥺";

    display.classList.remove("shake");

    void display.offsetWidth;

    display.classList.add("shake");

    setTimeout(() => {

      keypadClear();

    }, 700);

  }

}

// รองรับการกดตัวเลขจากคีย์บอร์ดด้วย

document.addEventListener("keydown", function(event) {

  const page2 = document.getElementById("page2");

  if(!page2.classList.contains("active")) return;

  if(/^[0-9]$/.test(event.key)) {

    keypadPress(event.key);

  } else if(event.key === "Backspace") {

    keypadDelete();

  } else if(event.key === "Escape") {

    keypadClear();

  } else if(event.key === "Enter") {

    keypadSubmit();

  }

});

function floatingHearts() {

  const symbols = [
    "♡","💗","💕","🌸","✿","✨"
  ];

  const item = document.createElement("div");

  item.className = "float";

  item.textContent =
    symbols[Math.floor(Math.random() * symbols.length)];

  item.style.left = Math.random() * 100 + "vw";

  item.style.fontSize =
    (14 + Math.random() * 20) + "px";

  item.style.animationDuration =
    (6 + Math.random() * 5) + "s";

  document.body.appendChild(item);

  setTimeout(() => item.remove(), 12000);

}

setInterval(floatingHearts, 1100);

function confetti() {

  const symbols = ["♡","💗","✨","🌸","💕"];

  for(let i = 0; i < 45; i++) {

    const item = document.createElement("div");

    item.className = "confetti";

    item.textContent =
      symbols[Math.floor(Math.random() * symbols.length)];

    item.style.left = Math.random() * 100 + "vw";

    item.style.top = "-30px";

    item.style.animationDelay =
      Math.random() * .8 + "s";

    document.body.appendChild(item);

    setTimeout(() => item.remove(), 5000);

  }

}
